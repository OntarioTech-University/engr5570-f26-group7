+def _fill_dp_level(dp, k, n):
+    prev = dp[k - 1]
+    curr = dp[k]

+        curr[i] = prev[prev[i]]

+def _build_dp_table(next_word, n, r):
+    max_log_r = r.bit_length()
+    dp = [[0] * (n + 1) for _ in range(max_log_r)]
+    dp[0] = next_word

+def _words_covered(dp, start, r):
