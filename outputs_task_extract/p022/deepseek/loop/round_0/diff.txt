+def shrink_window(i, j, current_len, word_lens):
+    """Remove word i-1 from the current window and reset if needed."""
+    if i > 0:
+        current_len -= word_lens[i - 1]
+        if j > i:

+def expand_window(j, current_len, i, word_lens, c, n):
+    """Expand the window as far as possible starting at word i."""
+    while j < n:
+        space = 1 if j > i else 0
+        if current_len + space + word_lens[j] <= c:
+            current_len += space + word_lens[j]
+            j += 1
+        else:
+            break
+    return j, current_len

+def precompute_next_word(n, c, word_lens):
+    """Compute next_word[i] for all i using a two-pointer approach."""
+    next_word = [0] * (n + 1)
+    j = 0
+    current_len = 0
+
+    for i in range(n):
+        j, current_len = shrink_window(i, j, current_len, word_lens)
+        j, current_len = expand_window(j, current_len, i, word_lens, c, n)
+        next_word[i] = j
+
+    next_word[n] = n
+    return next_word

+def build_dp(next_word, n, r):
+    """Build binary lifting table dp[k][i]."""
+    max_log_r = r.bit_length()
+    dp = [[0] * (n + 1) for _ in range(max_log_r)]
+    dp[0] = next_word
+
+    for k in range(1, max_log_r):
+        for i in range(n + 1):
+            dp[k][i] = dp[k - 1][dp[k - 1][i]]
+
+    return dp

+def find_best_start(n, r, dp):
