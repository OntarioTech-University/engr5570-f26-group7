    # Helper function to test feasibility of mission with given capacity
    def can_do_mission(start, end, cap):
        visited = {}  # node -> max_energy_when_visited
        queue = deque()
        queue.append((start, cap))  # node, energy
        visited[start] = cap

        while queue:
            node, energy = queue.popleft()
            if node == end:
                return True
            for nei, weight in adj[node]:
                if weight > energy:
                    continue
                new_energy = energy - weight
                if nei in centrals:
                    new_energy = cap
                if nei not in visited or visited[nei] < new_energy:
                    visited[nei] = new_energy
                    queue.append((nei, new_energy))
        return False

    results = []

    # For each query, binary search on capacity
    INF = 1 << 40
    for a, b in queries:
        low = 0
        high = INF
        res = -1
        while low <= high:

            if can_do_mission(a, b, mid):
