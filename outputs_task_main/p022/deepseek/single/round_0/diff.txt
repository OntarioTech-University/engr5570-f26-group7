+def _build_dp_table(n, r, next_word):
+    """Build binary lifting table for 2^k line jumps."""
+    max_log_r = r.bit_length()
+    dp = [[0] * (n + 1) for _ in range(max_log_r)]

+

+        curr = dp[k]
+        for i in range(n + 1):
+            curr[i] = prev[prev[i]]
+
+    return dp, max_log_r
